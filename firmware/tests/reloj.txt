# El MIDI se lee al principio de cada bloque de audio (5.8 ms): después de cada mensaje suelto va un wait 10.
# Reloj MIDI de afuera: tempo, Start/Stop/Continue, secuenciador y arpegiador (firmware/tests/correr.sh)
wait 600
expect externo 0
# 48 pulsos a 21 ms = 119 BPM: el aparato los sigue.
clock 48 21
expect externo 1
expect bpm_externo >115
expect bpm_externo <123
# Secuenciador: PLAY, luego Start; el primer pulso da el paso 1 y cada 6 pulsos el siguiente.
tap PLAY
expect tocando 1
midi FA 00 00
wait 10
clock 1 21
expect paso 0
clock 5 21
expect paso 0
clock 1 21
expect paso 1
clock 6 21
expect paso 2
# Stop detiene el avance aunque sigan los pulsos; Continue lo retoma.
midi FC 00 00
wait 10
expect transporte 0
clock 12 21
expect paso 2
midi FB 00 00
wait 10
expect transporte 1
clock 6 21
expect paso 3
# El contador de la celda Clock: compás y tiempo desde el Start.
tap SEQ
wait 40
shot reloj-contador.png
# A 60 BPM (42 ms) el tempo baja.
clock 48 42
expect bpm_externo >57
expect bpm_externo <63
# Sin pulsos medio segundo: vuelve al tempo propio y, como seguía corriendo, no se detiene.
wait 700
expect externo 0
tap PLAY
expect tocando 0
# Arpegiador sincronizado (Arp Pluck, semicorcheas): con el reloj a 119 y luego a 60 BPM.
